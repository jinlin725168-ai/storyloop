# 子项目 2a：纠偏 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 每轮失败都标明该谁改：判据与合同的问题第一次出现就停给人，环境问题不冒充实现失败，同一问题反复修不好时停在 `NO_PROGRESS`；所有停点输出统一的停止报告；合同缺陷可以作废并修订重开。

**Architecture:** 新增五个小模块，各管一件事：`failure_attribution`（发现校验、分流、失败特征、无进展判定，纯函数）、`project_loop_failures`（把失败写进运行状态并按归属停下或驳回）、`stop_report`（停止报告的构造与渲染）、`facts`（事实记录）、`supersede`（作废并修订重开）、`semantic_repair`（语义校验失败时带错误重试一次）。`project_loop._run_attempt` 只改调用点，状态机结构不变。

**Tech Stack:** Python 3.11 标准库；测试用 `unittest`，命令 `.venv/bin/python -m unittest ...`，从仓库根目录运行。

**Spec:** `docs/superpowers/specs/2026-10-14-子项目2a-纠偏-design.md`（Q1–Q8 均按推荐）

## Review Focus

1. **只有文字、没有位置的审查发现**：特征里只记 `target|criterion|文件`，两轮内容不同但都没给位置的发现会算成同一特征，可能过早进入根因调查。期望：没有 `criterion_id` 和 `location` 的发现，特征用 `perspective` 区分，仍不比较文字（Task 1 的 `test_findings_without_locations_are_told_apart_by_perspective`）。
2. **驳回后同一发现再次出现**：人已驳回，审查再提同一个判据问题时不能再停。期望：按实现处理并注明"已被人驳回"（Task 3 的 `test_an_overruled_finding_does_not_stop_again`）。
3. **在审查判定的 `ORACLE_DEFECT` 上直接 `run`**：不带 `--overrule` 时不能悄悄重证 RED 后继续实现。期望：报错并指出两条路（Task 3 的 `test_plain_run_on_a_review_routed_stop_is_refused`）。
4. **`acceptance-repair` 之后的无进展计数**：修复前的轮次不能和修复后的混在一起判定。期望：按 `epoch` 分开计数（Task 1 的 `test_counting_restarts_after_an_acceptance_repair`）。
5. **在旧功能分支上执行 supersede 后直接 approve 新变更**：新功能分支不能从旧功能分支上切出。期望：新运行开始前要求当前分支是旧运行的源分支（Task 7 的 `test_the_new_run_must_start_from_the_old_source_branch`）。

```python
"""Who has to fix a failed attempt, and whether the loop is still making progress.

The engine decides what it can observe (a layer red only with environment errors is `environment`);
the review proposes a target for each finding and the engine checks the evidence before believing it
(subproject 2a, decisions Q1 and Q5). Signatures use only stable identifiers, never message text.
"""

from __future__ import annotations

import hashlib
import re
from typing import Callable

TARGETS = ("implementation", "oracle", "contract")
LAYER_NAMES = ("unit", "integration", "acceptance", "smoke", "checks")
LOCATION = re.compile(r"^(?P<path>[^:]+?)(?::\d+)*$")


def contract_ids(change: dict) -> set[str]:
    stories = change["product"]["stories"]
    return {story["id"] for story in stories} | {c["id"] for story in stories for c in story["acceptance"]}


def location_file(location: object) -> str | None:
    """The repository-relative file of a `path:line` location, or None when it is not one."""
    if not isinstance(location, str):
        return None
    match = LOCATION.match(location.strip().replace("\\", "/"))
    if not match:
        return None
    path = match.group("path").strip()
    if not path or path.startswith("/") or ".." in path.split("/"):
        return None
    return path


def finding_key(item: dict) -> str:
    return f"{item.get('criterion_id') or '-'}|{location_file(item.get('location')) or '-'}"


def check_findings(findings: list, change: dict, oracle_paths: list[str],
                   exists: Callable[[str], bool], overruled: set[str]) -> list[dict]:
    """Copies of the findings, each with a target the evidence supports."""
    ids = contract_ids(change)
    checked = []
    for finding in findings:
        item = dict(finding) if isinstance(finding, dict) else {"message": str(finding)}
        target = item.get("target") or "implementation"
        if target != "implementation" and finding_key(item) in overruled:
            item["attribution_note"] = "已被人驳回，按实现问题处理"
            target = "implementation"
        problem = _evidence_problem(item, target, ids, oracle_paths, exists)
        if problem:
            item["attribution_note"] = f"归属证据不足（{problem}），按实现问题处理"
            target = "implementation"
        item["target"] = target
        checked.append(item)
    return checked


def _evidence_problem(item: dict, target: str, ids: set[str], oracle_paths: list[str],
                      exists: Callable[[str], bool]) -> str | None:
    if target == "implementation":
        return None
    if target not in TARGETS:
        return f"未知的 target {target}"
    if item.get("criterion_id") not in ids:
        return "criterion_id 不在合同里"
    path = location_file(item.get("location"))
    if path is None or not exists(path):
        return "location 指向的文件不存在"
    if target == "oracle" and not any(path == root or path.startswith(root.rstrip("/") + "/")
                                      for root in oracle_paths):
        return "location 不在验收套件或特征化测试里"
    return None


def route(findings: list[dict]) -> str:
    """A wrong contract invalidates oracle and code alike; a wrong oracle cannot be satisfied by code."""
    targets = {item.get("target") for item in findings}
    for target in ("contract", "oracle"):
        if target in targets:
            return target
    return "implementation"


def layer_attribution(layers: dict) -> str:
    red = [layers[name] for name in LAYER_NAMES
           if isinstance(layers.get(name), dict) and layers[name].get("status") not in {"PASSED", "SKIPPED"}]
    bad = [result for layer in red for result in layer.get("results") or []
           if result.get("status") not in {"PASS", "NOT_RUN"}]
    if red and bad and all(result.get("status") == "ERROR" for result in bad) \
            and not any(layer.get("oracle_problems") for layer in red):
        return "environment"
    return "implement"


def failure_signature(failure: dict) -> str:
    parts: set[str] = set()
    for name in LAYER_NAMES:
        layer = failure.get(name)
        if isinstance(layer, dict) and layer.get("status") not in {"PASSED", "SKIPPED"}:
            parts.add(f"layer:{name}:{layer.get('status')}")
            parts.update(f"case:{name}:{r.get('id')}:{r.get('status')}"
                         for r in layer.get("results") or [] if r.get("status") != "PASS")
    if "characterization" in failure:
        parts.update(f"lock:{path}" for path in failure["characterization"].get("uncharacterized") or [])
    if "review" in failure:
        findings = [f for f in (failure["review"] or {}).get("findings") or [] if isinstance(f, dict)]
        parts.update(f"review:{f.get('target') or 'implementation'}:{finding_key(f)}:{f.get('perspective') or '-'}"
                     for f in findings)
        if not findings:
            parts.add("review:no-findings")
    if "contract" in failure:
        parts.add("contract:INVALID")
    return hashlib.sha256("\n".join(sorted(parts)).encode("utf-8")).hexdigest()


def _counted(failures: list[dict], epoch: int) -> list[dict]:
    return [item for item in failures if item.get("signature") and int(item.get("epoch", 0)) == epoch]


def next_mode(failures: list[dict], epoch: int) -> str | None:
    """Two attempts in a row with the same signature: the next one investigates the root cause first."""
    recent = _counted(failures, epoch)[-2:]
    if len(recent) == 2 and recent[0]["signature"] == recent[1]["signature"]:
        return "root_cause"
    return None


def no_progress(failures: list[dict], epoch: int) -> bool:
    """The root-cause attempt changed nothing: same signature (so no failing case turned green)."""
    recent = _counted(failures, epoch)
    return (len(recent) >= 2 and recent[-1].get("mode") == "root_cause"
            and recent[-1]["signature"] == recent[-2]["signature"])
```

## 判据与合同的归属（review 使用）

审查发现要标出该谁改：

- `implementation`：代码没做到合同要求。默认值。
- `oracle`：验收用例或特征化测试本身写错了。例如断言与合同不符、依赖候选副本里没有的东西（`.git`、网络、本机路径）、引用了本次变更才新建的文件、前提不满足时降级通过。`location` 指向那个测试文件的行。
- `contract`：按合同做不出来。例如两条验收条件互相矛盾，或者某条条件与必须保持的既有行为冲突。`location` 指向与合同矛盾的代码事实，`criterion_id` 写出冲突的条件。

```python
"""Recording failed attempts and routing them to whoever has to fix them (subproject 2a §3, §5)."""

from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from .failure_attribution import (check_findings, failure_signature, finding_key, no_progress, route)
from .models import require, text
from .project_loop_support import pause, save_run
from .smoke import suite_relative

ROUTED_STOPS = {"oracle": "ORACLE_DEFECT", "contract": "NEEDS_SPEC_FIX"}


def epoch_of(state: dict) -> int:
    """Each acceptance-repair voids earlier attempts; failures are counted per epoch."""
    return len(state.get("repairs") or [])


def _append(state: dict, entry: dict, attribution: str) -> None:
    item = {**entry, "attribution": attribution, "epoch": epoch_of(state), "mode": state.get("mode")}
    item["signature"] = failure_signature(item)
    state["failures"].append(item)


def record_failure(project: Path, state: dict, entry: dict, attribution: str) -> dict | None:
    """Keep the failed attempt; the next attempt gets it. Returns a stop when the loop stopped progressing."""
    _append(state, entry, attribution)
    save_run(project, state)
    if no_progress(state["failures"], epoch_of(state)):
        return pause(project, state, "NO_PROGRESS",
                     error="根因调查后失败特征仍与上一轮相同，且没有原先失败的用例转绿")
    return None


def oracle_paths(project: Path, state: dict) -> list[str]:
    locked = (state.get("characterization") or {}).get("tests") or []
    return [suite_relative(project, state["change_id"]), *locked]


def route_review(project: Path, change: dict, state: dict, review: dict) -> dict | None:
    """A failed review goes back to the implementer unless its checked findings blame the oracle or contract."""
    overruled = set(state.get("overruled") or [])
    findings = check_findings(review.get("findings") or [], change, oracle_paths(project, state),
                              lambda path: (project / path).is_file(), overruled)
    review = {**review, "findings": findings}
    entry = {"attempt": state["attempt"], "review": review}
    target = route(findings)
    if target == "implementation":
        return record_failure(project, state, entry, "implement")
    _append(state, entry, target)
    blamed = [item for item in findings if item["target"] == target]
    details = {"routed": {"source": "review", "target": target, "findings": blamed},
               "error": blamed[0].get("message", "")}
    if target == "oracle":
        details["acceptance"] = {"status": "ORACLE_DEFECT", "source": "review", "findings": blamed}
    return pause(project, state, ROUTED_STOPS[target], **details)


def require_routed_choice(state: dict, overrule: str | None) -> None:
    """A stop the review routed waits for a human; only an explicit overrule continues implementing."""
    routed = state.get("status") in ROUTED_STOPS.values() and state.get("routed")
    if routed and not overrule:
        require(False, f"{state['status']}：审查判定问题不在实现。修正判据用 storyloop acceptance-repair "
                       "--reason <原因>；合同缺陷用 storyloop change --supersede；"
                       "不同意这个判定用 storyloop run --overrule <理由>")
    require(not overrule or routed, "--overrule 只能用于审查判定的 ORACLE_DEFECT 或 NEEDS_SPEC_FIX")


def apply_overrule(project: Path, state: dict, reason: str) -> None:
    """The human disagrees with the review: its blamed findings go to the implementer, on the record."""
    reason = text(reason, "overrule reason")
    routed = state.pop("routed")
    keys = sorted({finding_key(item) for item in routed["findings"]})
    state.setdefault("overrules", []).append({
        "at": datetime.now(timezone.utc).isoformat(timespec="seconds"), "status": state["status"],
        "attempt": state["attempt"], "reason": reason, "findings": keys})
    state["overruled"] = sorted(set(state.get("overruled") or []) | set(keys))
    last = state["failures"][-1] if state.get("failures") else {}
    if last.get("attribution") in ROUTED_STOPS:
        last.update({"attribution": "implement", "overruled": True})
    for key in ("acceptance", "error", "questions", "next", "stop"):
        state.pop(key, None)
    state["status"] = "IMPLEMENTING"
    save_run(project, state)
```

            self.assertEqual(set(stop), KEYS, status)

        self.assertFalse(any("--overrule" in o["command"] for o in plain["options"]))

"""One shape for every stop: what blocked, the evidence, the human's options and how to resume (2a §6)."""

from __future__ import annotations

from pathlib import Path

LAYER_NAMES = ("unit", "integration", "acceptance", "smoke", "checks")
PHASES = {
    "NEEDS_INPUT": "implement", "ORACLE_DEFECT": "prepare", "EXISTING_GREEN": "prepare",
    "CHARACTERIZATION_FAILED": "prepare", "BOUNDARY_VIOLATION": "implement", "BASE_CHANGE_REQUIRED": "implement",
    "WORKSPACE_TAINTED": "implement", "BUDGET_EXHAUSTED": "implement", "NO_PROGRESS": "implement",
    "MANUAL_ACCEPTANCE_REQUIRED": "verify", "NEEDS_SPEC_FIX": "verify",
}
STOP_STATUSES = frozenset(PHASES)
BLOCKERS = {
    "NEEDS_INPUT": "代理提出了需要人决定的问题",
    "ORACLE_DEFECT": "验收用例本身有缺陷，实现无法让它正确地通过",
    "EXISTING_GREEN": "验收用例在实现前就已通过，证明不了 RED",
    "CHARACTERIZATION_FAILED": "特征化测试在旧代码上不绿",
    "BOUNDARY_VIOLATION": "代理写了不允许的路径",
    "BASE_CHANGE_REQUIRED": "改动碰到了底座",
    "WORKSPACE_TAINTED": "工作区出现了非预期的改动",
    "BUDGET_EXHAUSTED": "实现轮次用完了",
    "NO_PROGRESS": "根因调查后失败特征仍不变",
    "MANUAL_ACCEPTANCE_REQUIRED": "有验收条件需要人工核对",
    "NEEDS_SPEC_FIX": "合同按现在的写法做不出来",
}


def _options(status: str, where: str, project: str, change_id: str, routed: bool) -> list[dict]:
    repair = {"action": "修正判据", "command": f"storyloop acceptance-repair {where} --reason <原因>"}
    supersede = {"action": "作废并修订重开",
                 "command": f"storyloop change {project} --supersede {change_id} --message <修订意见>"
                            + ("" if status == "NEEDS_SPEC_FIX" else " --reason <为什么是合同的问题>")}
    overrule = {"action": "驳回归属", "command": f"storyloop run {where} --overrule <理由>"}
    table = {
        "NEEDS_INPUT": [{"action": "回答", "command": f"storyloop answer {where} --answer <回答>"}],
        "ORACLE_DEFECT": [repair, *([overrule] if routed else [])],
        "EXISTING_GREEN": [repair],
        "CHARACTERIZATION_FAILED": [{"action": "就地修正特征化测试后续跑", "command": f"storyloop run {where}"}],
        "BOUNDARY_VIOLATION": [{"action": "把需要的路径走一次底座提案",
                                "command": f"storyloop base-propose {project} --path <路径> --reason <原因>"}],
        "BASE_CHANGE_REQUIRED": [{"action": "独立提交底座改动后升级底座",
                                  "command": f"storyloop base-update {project} --proposal <BASE-ID>"}],
        "WORKSPACE_TAINTED": [{"action": "清理后复位", "command": f"storyloop reset {where} --reason <原因>"}],
        "BUDGET_EXHAUSTED": [repair, supersede],
        "NO_PROGRESS": [repair, supersede],
        "MANUAL_ACCEPTANCE_REQUIRED": [{"action": "逐条人工验收",
                                        "command": f"storyloop accept {where} --criterion <ID> --note <核对了什么>"}],
        "NEEDS_SPEC_FIX": [supersede, *([overrule] if routed else [])],
    }
    return table.get(status, [])


def failure_view(failure: dict) -> dict:
    """One line per failed attempt: which gate said no and the gist of why."""
    attempt = failure.get("attempt")
    if "contract" in failure:
        return {"attempt": attempt, "gate": "contract", "reason": failure["contract"].get("message", "")}
    if "review" in failure:
        review = failure["review"] or {}
        findings = review.get("findings") or []
        first = findings[0].get("message", "") if findings and isinstance(findings[0], dict) else ""
        return {"attempt": attempt, "gate": "review",
                "reason": f"{len(findings)} finding(s): {first or review.get('summary', '')}"[:300]}
    if "characterization" in failure:
        paths = failure["characterization"].get("uncharacterized") or []
        return {"attempt": attempt, "gate": "characterization",
                "reason": "modified without a lock, restored and locked: " + ", ".join(paths[:5])}
    red = [name for name in LAYER_NAMES
           if isinstance(failure.get(name), dict) and failure[name].get("status") not in {"PASSED", "SKIPPED"}]
    if red:
        return {"attempt": attempt, "gate": "layers",
                "reason": ", ".join(f"{name}={failure[name].get('status')}" for name in red)}
    return {"attempt": attempt, "gate": "unknown", "reason": ""}


def _completed(state: dict) -> list[str]:
    done = []
    if state.get("contract_commit"):
        done.append("contract")
    if state.get("red_evidence_sha256"):
        done.append("acceptance_red")
    if (state.get("design_plan") or {}).get("status") in {"DONE", "SKIPPED"}:
        done.append("design")
    if (state.get("characterization") or {}).get("status") in {"GREEN", "SKIPPED"}:
        done.append("characterization")
    done += [f"attempt:{item.get('attempt')}" for item in state.get("failures") or []]
    return done


def _evidence(state: dict, details: dict) -> list[str]:
    routed = (state.get("routed") or {}).get("findings") or []
    found = [item["location"] for item in routed if item.get("location")]
    found += [item.get("path", "") for item in details.get("violations") or state.get("violations") or []]
    acceptance = details.get("acceptance") or state.get("acceptance") or {}
    found += [f"条件 {cid}" for cid in acceptance.get("criteria_failing") or acceptance.get("criteria_green") or []]
    found += [f"条件 {item.get('criterion_id') or item.get('id')}" if isinstance(item, dict) else f"条件 {item}"
              for item in details.get("pending") or []]
    return [item for item in dict.fromkeys(found) if item]


def build_stop(project: Path | str, state: dict, status: str, details: dict) -> dict:
    change_id = state["change_id"]
    where = f"{project} --change {change_id}"
    routed = bool(state.get("routed"))
    failures = state.get("failures") or []
    last = failure_view(failures[-1]) if failures else None
    return {
        "status": status,
        "phase": "verify" if routed else PHASES.get(status, "implement"),
        "completed": _completed(state),
        "last_verdict": {"gate": last["gate"], "summary": last["reason"]} if last else None,
        "attempts": [{"attempt": item.get("attempt"), "attribution": item.get("attribution", "implement"),
                      "signature": (item.get("signature") or "")[:8], "summary": failure_view(item)["reason"]}
                     for item in failures],
        "blocker": details.get("error") or state.get("error") or BLOCKERS.get(status, status),
        "evidence": _evidence(state, details),
        "facts": list(state.get("facts") or []),
        "questions": list(details.get("questions") or state.get("questions") or []),
        "options": _options(status, where, str(project), change_id, routed),
        "resume": f"storyloop run {where}",
    }


def stop_of(project: Path | str, run: dict | None) -> dict | None:
    """The saved stop, or one built now for a run saved before stops had reports."""
    if not run or run.get("status") not in STOP_STATUSES:
        return None
    return run.get("stop") or build_stop(project, run, run["status"], {})


def render_stop(stop: dict) -> list[str]:
    lines = [f"阻塞点：{stop['blocker']}"]
    if stop["evidence"]:
        lines.append("证据：")
        lines += [f"  - {item}" for item in stop["evidence"]]
    if stop["questions"]:
        lines.append("问题：")
        lines += [f"  - {item}" for item in stop["questions"]]
    if stop["options"]:
        lines.append("可选操作：")
        lines += [f"  {n}. {o['action']}：{o['command']}" for n, o in enumerate(stop["options"], 1)]
    lines.append(f"恢复：{stop['resume']}")
    return lines
```

"""The run's fact record: what agents found in the code, each with a location the engine checked (2a Q7).

Facts are agent statements, not machine facts; the engine only checks that the cited file exists. They
travel to every later call of the run and into stop reports and contract revisions.
"""

from __future__ import annotations

from pathlib import Path

from .failure_attribution import location_file

MAX_FACTS = 50


def merge_facts(project: Path, state: dict, facts: object, role: str) -> None:
    if not isinstance(facts, list):
        return
    record = list(state.get("facts") or [])
    known = {(item["statement"], location_file(item["location"])) for item in record}
    for item in facts:
        if not isinstance(item, dict):
            continue
        statement, location = item.get("statement"), item.get("location")
        path = location_file(location)
        if not (isinstance(statement, str) and statement.strip() and len(statement) <= 500 and path
                and (Path(project) / path).is_file()):
            continue
        key = (statement.strip(), path)
        if key in known:
            continue
        known.add(key)
        record.append({"statement": statement.strip(), "location": location.strip(), "role": role,
                       "attempt": state.get("attempt")})
    state["facts"] = record[-MAX_FACTS:]
```

```python
"""`change --supersede`: void a run whose contract is wrong and reopen the change, revised and linked (2a §4)."""

from __future__ import annotations

from copy import deepcopy
from pathlib import Path
from typing import Callable

from .changes import _save, create_change, load_change
from .models import identifier, require, text
from .project_loop_support import save_run
from .stop_report import stop_of
from .storage import load_json

SUPERSEDABLE = frozenset({"NEEDS_SPEC_FIX", "BUDGET_EXHAUSTED", "NO_PROGRESS", "ORACLE_DEFECT",
                          "EXISTING_GREEN", "NEEDS_INPUT", "IMPLEMENTING", "BOUNDARY_VIOLATION",
                          "CHARACTERIZATION_FAILED"})


def supersede_change(project: Path, old_id: str, message: str, reviser: Callable[[dict], dict],
                     reason: str | None = None) -> dict:
    """`reviser` is the product role: it gets the old contract, the evidence and the message."""
    project = Path(project).resolve()
    old_id = identifier(old_id, "change id")
    message = text(message, "revision message")
    old = load_change(project, old_id)
    run_file = project / ".storyloop/state/runs" / f"{old_id}.json"
    require(run_file.exists(), "Only a started run can be superseded; before that, revise the draft")
    run = load_json(run_file)
    require(run.get("status") in SUPERSEDABLE,
            f"A run in {run.get('status')} cannot be superseded; a merged change is revised by a new change")
    if run["status"] != "NEEDS_SPEC_FIX":
        require(reason is not None, "Superseding outside NEEDS_SPEC_FIX needs --reason: why is the contract wrong")
        reason = text(reason, "supersede reason")
    stop = stop_of(project, run) or {}
    plan = run.get("design_plan") or {}
    link = {"change_id": old_id, "message": message, "reason": reason, "blocker": stop.get("blocker"),
            "source_branch": run["source_branch"], "design_summary": plan.get("summary")}
    product = reviser({"idea": old["idea"], "message": message, "current_product": old["product"],
                       "supersede": {**link, "facts": run.get("facts") or [], "evidence": stop.get("evidence") or []}})
    new = create_change(project, old["idea"], product)
    new["supersedes"] = link
    _save(project, new)
    old_state = {**old, "status": "SUPERSEDED", "superseded_by": new["change_id"]}
    _save(project, old_state)
    run.update({"status": "SUPERSEDED", "superseded_by": new["change_id"]})
    save_run(project, run)
    result = deepcopy(new)
    result["next"] = (f"旧分支 feature/{old_id} 保留：先提交或暂存其中未提交的改动，git checkout {run['source_branch']}，"
                      f"再 storyloop approve {project} --change {new['change_id']}")
    return result
```

（`_save` 是 `changes.py` 的模块内函数，这里按现有约定直接复用，不新增公共接口。）

```python
"""One in-session repair for answers that pass the role schema but fail the engine's semantic checks.

Schema errors are repaired inside the adapters; a semantic error (a reproduces_defect on a must-hold row,
a design naming a file that does not exist) used to void the whole call. The role now gets its previous
answer and the errors once; a second failure stops as before (subproject 2a §7).
"""

from __future__ import annotations

from typing import Callable

from .models import StoryLoopError

REPAIRABLE = {"BLOCKED"}  # the code `require` raises for validation failures


def answer_with_repair(call: Callable[[dict], object], request: dict,
                       validate: Callable[[object], None]) -> object:
    answer = call(request)
    try:
        validate(answer)
        return answer
    except StoryLoopError as exc:
        if exc.code not in REPAIRABLE:
            raise
        retry = call({**request, "previous_response": answer, "validation_errors": [str(exc)]})
    validate(retry)
    return retry
```

```python
def correction_view(state: dict, change: dict) -> dict:
    from .stop_report import failure_view  # local import: the report module is imported by the loop
    return {"failures": [{"attempt": item.get("attempt"), "attribution": item.get("attribution", "implement"),
                          "signature": (item.get("signature") or "")[:8], **failure_view(item)}
                         for item in state.get("failures") or []],
            "overrules": list(state.get("overrules") or []),
            "supersedes": (change.get("supersedes") or {}).get("change_id")}


def _correction_markdown(view: dict) -> list[str]:
    if not view["failures"] and not view["overrules"] and not view["supersedes"]:
        return []
    lines = ["", "## 失败归属", ""]
    if view["supersedes"]:
        lines += [f"修订自 {view['supersedes']}（作废并修订重开）", ""]
    if view["failures"] or view["overrules"]:
        lines += ["| 轮次 | 归属 | 特征 | 说明 |", "|---|---|---|---|"]
        lines += [f"| {item['attempt']} | {item['attribution']} | {item['signature'] or '-'} | "
                  f"{_markdown(item['gate'] + ': ' + item['reason'][:120])} |" for item in view["failures"]]
        lines += [f"| {item.get('attempt')} | 人工驳回 | - | {_markdown(item.get('reason', ''))} |"
                  for item in view["overrules"]]
    return lines
```

`generate_change_report` 签名加 `correction: dict | None = None`，`report` 字典加 `"correction": correction or {"failures": [], "overrules": [], "supersedes": None},`；`_render_markdown` 在 `lines.extend(_oracle_markdown(...))` 之后加：
